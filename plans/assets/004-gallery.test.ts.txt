import { describe, expect, it } from "vitest";
import { PRESET_CATALOG } from "@wavr/core/presets";
import {
  GALLERY_BLENDED_KEYS, GALLERY_FADE_SECONDS, GALLERY_HOLD_SECONDS, GALLERY_ROOM_IDS, GALLERY_SKIP_SECONDS,
  blendRooms, easePassage, galleryFrame, galleryRooms, passage, skipWalk, startWalk, stepWalk,
} from "./gallery";

const rooms = galleryRooms();

describe("gallery programme", () => {
  it("shows every listed room the catalogue has, in order, and at least eight", () => {
    const known = GALLERY_ROOM_IDS.filter((id) => PRESET_CATALOG.some((preset) => preset.id === id));
    expect(rooms.map((room) => room.id)).toEqual(known);
    expect(rooms.length).toBeGreaterThanOrEqual(8);
    expect(new Set(GALLERY_ROOM_IDS).size).toBe(GALLERY_ROOM_IDS.length);
  });

  it("only holds rooms a passage can blend: one layer each, and the same unblended globals", () => {
    const skip = new Set<string>(["layers", "realBloomEnabled", ...GALLERY_BLENDED_KEYS]);
    const rest = (room: (typeof rooms)[number]) =>
      JSON.stringify(Object.entries(room.state).filter(([key]) => !skip.has(key)));
    for (const room of rooms) {
      expect(room.state.layers, room.id).toHaveLength(1);
      expect(rest(room), room.id).toBe(rest(rooms[0]));
    }
  });
});

describe("gallery walk", () => {
  it("rests, passes to the next room by itself, and wraps round", () => {
    let walk = startWalk(0);
    expect(stepWalk(walk, GALLERY_HOLD_SECONDS - 1, 3, true)).toBe(walk);
    walk = stepWalk(walk, GALLERY_HOLD_SECONDS, 3, true);
    expect(walk).toMatchObject({ room: 0, next: 1 });
    expect(passage(walk, GALLERY_HOLD_SECONDS + GALLERY_FADE_SECONDS / 2)).toBeCloseTo(0.5);
    walk = stepWalk(walk, GALLERY_HOLD_SECONDS + GALLERY_FADE_SECONDS, 3, true);
    expect(walk).toMatchObject({ room: 1, next: null });
    walk = startWalk(0, 2);
    expect(stepWalk(walk, GALLERY_HOLD_SECONDS, 3, true)).toMatchObject({ room: 2, next: 0 });
  });

  it("stays put when it may not advance, or has one room", () => {
    const walk = startWalk(0);
    expect(stepWalk(walk, 1e6, 3, false)).toBe(walk);
    expect(stepWalk(walk, 1e6, 1, true)).toBe(walk);
    expect(skipWalk(walk, 5, 1, 1)).toBe(walk);
  });

  it("a skip at rest begins a quick passage, backwards too", () => {
    expect(skipWalk(startWalk(0), 5, 3, 1)).toMatchObject({ room: 0, next: 1, since: 5, fadeSeconds: GALLERY_SKIP_SECONDS });
    expect(skipWalk(startWalk(0), 5, 3, -1)).toMatchObject({ room: 0, next: 2 });
  });

  it("a skip during a passage hurries it without a jump", () => {
    const slow = stepWalk(startWalk(0), GALLERY_HOLD_SECONDS, 3, true);
    const now = GALLERY_HOLD_SECONDS + GALLERY_FADE_SECONDS * 0.25;
    const hurried = skipWalk(slow, now, 3, 1);
    expect(hurried.next).toBe(1);
    expect(passage(hurried, now)).toBeCloseTo(passage(slow, now));
    expect(passage(hurried, now + GALLERY_SKIP_SECONDS * 0.75)).toBeCloseTo(1);
    expect(skipWalk(hurried, now + 1, 3, 1)).toBe(hurried);
  });
});

describe("gallery frame", () => {
  const [a, b] = [rooms[0].state, rooms[1].state];

  it("is the room itself at either end of a passage", () => {
    expect(blendRooms(a, b, 0)).toBe(a);
    expect(blendRooms(a, b, 1)).toBe(b);
  });

  it("lays the entered room over the one being left, at the passage's opacity", () => {
    const frame = blendRooms(a, b, 0.25);
    expect(frame.layers).toHaveLength(2);
    expect(frame.layers[0]).toBe(a.layers[0]);
    expect(frame.layers[1]).toMatchObject({ gradientType: b.layers[0].gradientType, opacity: 0.25 });
    expect(frame.grain).toBeCloseTo(a.grain + (b.grain - a.grain) * 0.25);
  });

  it("fades the bloom by amount instead of switching it", () => {
    const withBloom = rooms.find((room) => room.state.realBloomEnabled)!.state;
    const without = rooms.find((room) => !room.state.realBloomEnabled)!.state;
    expect(blendRooms(without, withBloom, 0.25)).toMatchObject({ realBloomEnabled: true, realBloomAmount: 0.25 });
    expect(blendRooms(withBloom, without, 0.25)).toMatchObject({ realBloomEnabled: true, realBloomAmount: 0.75 });
    expect(blendRooms(without, without, 0.5).realBloomEnabled).toBe(false);
  });

  it("rises out of the dark, ignores the pointer, and can stand still", () => {
    const walk = startWalk(0);
    expect(galleryFrame(rooms, walk, 0, 0, true).brightness).toBe(0);
    expect(galleryFrame(rooms, walk, 60, 0, true).brightness).toBe(a.brightness);
    expect(galleryFrame(rooms, walk, 60, 0, false)).toMatchObject({ mouseReact: 0, playing: false });
  });

  it("eases a passage in and out", () => {
    expect([easePassage(0), easePassage(0.5), easePassage(1)]).toEqual([0, 0.5, 1]);
    expect(easePassage(0.1)).toBeLessThan(0.1);
  });
});
